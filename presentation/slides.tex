\begin{frame}{Our decision tree (NumPy)}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Split search}
    \begin{itemize}
      \item Sort each feature; thresholds between distinct neighbours
      \item Prefix counts / sums $\Rightarrow$ every threshold in $O(n)$
      \item Pick the largest gain\\[0.2em]
        $\Delta I = I(\text{parent}) - \sum_c \frac{N_c}{N} I(c)$
    \end{itemize}
    \textbf{Impurities}\\[0.2em]
    \small
    \begin{tabular}{@{}ll@{}}
      Gini & $1-\sum_k p_k^2$\\
      Entropy & $-\sum_k p_k\log_2 p_k$\\
      MSE (regression) & $\frac1N\sum_i(y_i-\bar y)^2$
    \end{tabular}
    \column{0.42\textwidth}
    \textbf{Stopping controls} (each tested)
    \begin{itemize}\small
      \item \texttt{max\_depth}, \texttt{min\_samples\_split}
      \item \texttt{min\_samples\_leaf}, \texttt{min\_impurity\_decrease}
    \end{itemize}
    \textbf{Leaves}: majority class / mean;\\ \texttt{predict\_proba} = leaf frequencies\\[0.4em]
    \textbf{Checks}: matches sklearn's splits on a tiny, fully determined example; impurity-decrease importances
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Our SVM: Pegasos sub-gradient + random Fourier features}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Objective} (hinge primal, no QP solver)
    \[
      J(\mathbf w,b)=\tfrac{\lambda}{2}\lVert\mathbf w\rVert^2+\tfrac1n\sum_i\max\bigl(0,1-y_i(\mathbf w^\top\mathbf x_i+b)\bigr)
    \]
    \textbf{Update} for a random row $i$, $\eta_t=\frac{1}{\lambda t}$ or constant:
    \[
      \mathbf w\leftarrow(1-\eta_t\lambda)\mathbf w+\eta_t y_i\mathbf x_i\cdot\mathbf 1[y_i f(\mathbf x_i)<1]
    \]
    \aside{Unregularised bias, projection onto $\lVert\mathbf w\rVert\le1/\sqrt\lambda$, standardised inputs.}
    \column{0.42\textwidth}
    \textbf{Diagnostics}
    \begin{itemize}\small
      \item Objective logged at 51 checkpoints
      \item $\mathbf w$, margin width $2/\lVert\mathbf w\rVert$
      \item Support vectors: $y_i f(\mathbf x_i)\le 1$
    \end{itemize}
    \textbf{Non-linearity: RFF}
    \[ z(\mathbf x)=\sqrt{2/D}\cos(\mathbf xW+\mathbf c),\ W\sim\mathcal N(0,2\gamma) \]
    \aside{Same linear solver; fixed cost $D$ instead of $O(n)$ kernel sums per step.}
  \end{columns}
\end{frame}

\begin{frame}{Experimental setup}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Selection on validation only}
    \begin{itemize}\small
      \item Tree: depth $\{2,4,8,12\}\times$ leaf $\{1,5,20\}$, Gini vs entropy
      \item SVM: 33 predeclared configs ($\lambda$, schedule, $T$, $\gamma$, $D$)
      \item Ridge (bonus): $\alpha\in\{0.1,\dots,1000\}$
      \item Test split scored \hi{once} with frozen settings
    \end{itemize}
    \textbf{Uncertainty}\\
    \small 5-fold CV on train+val (test excluded), refitting encoder, scaler and tier threshold per fold; mean $\pm$ std
    \column{0.47\textwidth}
    \textbf{Matched sklearn baselines}
    \begin{itemize}\small
      \item sklearn decision trees: same grid and criteria
      \item \texttt{SVC} linear / RBF in a \texttt{Pipeline} with \texttt{StandardScaler}
      \item Our loss averages the hinge, SVC sums it:\\
        $C = 1/(n_\text{train}\lambda)$
    \end{itemize}
    \textbf{Reproducibility}\\
    \small \texttt{python -m src.run\_all} rebuilds every number and figure; seeds fixed, deps pinned, data and source hashes in the manifest; 348 tests pass
  \end{columns}
\end{frame}

    \begin{itemize}\small

      \item Val RMSE 124.7k $\to$ \hi{97.2k}, MAE 48.3k $\to$ 34.0k, $R^2$ 0.82
      \item Val MAE keeps falling as trees are added
    \end{itemize}

    \textbf{Honest limitations}
    \begin{itemize}\small
      \item Asking prices, Baku residential only
      \item Test split already seen $\Rightarrow$ improvements reported on validation, not test
      \item CV = variability of frozen settings, not nested selection
      \item 446 exact duplicate feature rows remain (effect $<$1\% MAE)
      \item Single-run timings on one laptop
    \end{itemize}
